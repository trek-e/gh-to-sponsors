/**
 * Gets link target from SOCIAL_LINK_TARGET env var
 * @returns 'ghost' | 'github' | custom URL string
 */
export function getLinkTarget(): string {
  return process.env.SOCIAL_LINK_TARGET || 'ghost';
}

/**
 * Composes social post content by appending link to teaser
 *
 * @param state - Post state with teaser and optional Ghost result
 * @param linkTarget - 'ghost', 'github', or custom URL
 * @param fallbackUrl - Optional fallback URL if primary unavailable
 * @returns Composed text ready for social posting
 */
export function composeSocialPostContent(
  state: PostState,
  linkTarget: string,
  fallbackUrl?: string
): string {
  const teaserText = state.teaser?.text || '';
  if (!teaserText) return '';

  let url: string | undefined;

  if (linkTarget === 'ghost') {
    // Get Ghost URL from state if available and successful
    const ghostResult = state.platforms?.['ghost'];
    if (ghostResult?.status === 'success' && ghostResult.postUrl) {
      url = ghostResult.postUrl;
    } else if (fallbackUrl) {
      url = fallbackUrl;
    }
  } else if (linkTarget === 'github') {
    url = fallbackUrl;
  } else if (linkTarget.startsWith('http')) {
    // Custom URL
    url = linkTarget;
  }

  if (url) {
    return `${teaserText}\n\n${url}`;
  }
  return teaserText;
}

```typescript
export async function postToAllPlatforms(
  plugins: PlatformPlugin[],
  state: PostState,
  options?: { githubFallbackUrl?: string }
): Promise<ExecutionSummary> {
  // Determine posting order: Ghost first (to get URL), then others
  const ghostPlugin = plugins.find(p => p.name === 'ghost');
  const socialPlugins = plugins.filter(p => p.name !== 'ghost');

  const results: PlatformPostResult[] = [];
  let updatedState = { ...state };

  // Post to Ghost first if configured
  if (ghostPlugin) {
    try {
      const result = await ghostPlugin.post(state);
      const ghostResult: PlatformPostResult = {
        platform: ghostPlugin.name,
        success: result.success,
        postId: result.platformPostId,
        postUrl: result.platformUrl,
        error: result.error,
      };
      results.push(ghostResult);

      // Update state with Ghost result for social posts to reference
      if (result.success && result.platformUrl) {
        updatedState = {
          ...updatedState,
          platforms: {
            ...updatedState.platforms,
            ghost: {
              status: 'success',
              postUrl: result.platformUrl,
              postId: result.platformPostId,
              attemptedAt: new Date().toISOString()
            }
          }
        };
      }
    } catch (error) {
      results.push({
        platform: ghostPlugin.name,
        success: false,
        error: error instanceof Error ? error.message : String(error),
      });
    }
  }

  // Get link target and compose content for social platforms
  const linkTarget = getLinkTarget();

  // Create modified state with composed teaser for social posts
  const socialState = {
    ...updatedState,
    teaser: updatedState.teaser ? {
      ...updatedState.teaser,
      text: composeSocialPostContent(updatedState, linkTarget, options?.githubFallbackUrl)
    } : undefined
  };

  // Post to social platforms in parallel with composed content
  if (socialPlugins.length > 0) {
    const socialPromises = socialPlugins.map(async (plugin): Promise<PlatformPostResult> => {
      try {
        const result = await plugin.post(socialState);
        return {
          platform: plugin.name,
          success: result.success,
          postId: result.platformPostId,
          postUrl: result.platformUrl,
          error: result.error,
        };
      } catch (error) {
        return {
          platform: plugin.name,
          success: false,
          error: error instanceof Error ? error.message : String(error),
        };
      }
    });

    const socialSettled = await Promise.allSettled(socialPromises);
    for (const outcome of socialSettled) {
      if (outcome.status === 'fulfilled') {
        results.push(outcome.value);
      } else {
        results.push({
          platform: 'unknown',
          success: false,
          error: String(outcome.reason),
        });
      }
    }
  }

  const successfulPlatforms = results.filter((r) => r.success).map((r) => r.platform);
  const failedPlatforms = results.filter((r) => !r.success).map((r) => r.platform);

  return {
    allSucceeded: failedPlatforms.length === 0 && results.length > 0,
    anySucceeded: successfulPlatforms.length > 0,
    results,
    successfulPlatforms,
    failedPlatforms,
  };
}
```
