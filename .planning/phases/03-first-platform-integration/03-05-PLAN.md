/**
 * Generates a retry token for failed platform posts
 *
 * Creates a token that allows retrying specific platforms without
 * re-approving the entire digest.
 *
 * @param postId - Post identifier
 * @param platforms - Array of platform names to retry
 * @param ttlHours - Token time-to-live in hours
 * @param secret - HMAC secret key
 * @returns Signed token string
 */
export function generateRetryToken(
  postId: string,
  platforms: string[],
  ttlHours: number,
  secret: string
): string {
  const payload: TokenPayload = {
    postId,
    action: 'retry',
    platforms,
    exp: Date.now() + (ttlHours * 60 * 60 * 1000),
    jti: randomUUID()
  };

  return signToken(payload, secret);
}
```

export interface FailureNotificationData {
  postTitle: string;
  successfulPlatforms: Array<{ name: string; url?: string }>;
  failedPlatforms: Array<{ name: string; error: string; retryLink: string }>;
  expirationHours: number;
}

/**
 * Renders the failure notification email with retry links
 */
export function renderFailureNotificationEmail(data: FailureNotificationData): EmailTemplate {
  const allFailed = data.successfulPlatforms.length === 0;
  const subject = allFailed
    ? `Posting failed: ${data.postTitle}`
    : `Partial success: ${data.postTitle}`;

  const html = renderFailureHTML(data);
  const text = renderFailureText(data);

  return { html, text, subject };
}

function renderFailureHTML(data: FailureNotificationData): string {
  const successSection = data.successfulPlatforms.length > 0 ? `
    <tr>
      <td style="padding: 16px 32px;">
        <h3 style="margin: 0 0 12px; color: #1a7f37; font-size: 16px;">Succeeded</h3>
        <ul style="margin: 0; padding-left: 20px; color: #24292f;">
          ${data.successfulPlatforms.map(p => `
            <li style="margin-bottom: 8px;">
              <strong>${escapeHtml(p.name)}</strong>
              ${p.url ? ` - <a href="${escapeHtml(p.url)}" style="color: #0969da;">${escapeHtml(p.url)}</a>` : ''}
            </li>
          `).join('')}
        </ul>
      </td>
    </tr>
  ` : '';

  const failedSection = `
    <tr>
      <td style="padding: 16px 32px;">
        <h3 style="margin: 0 0 12px; color: #cf222e; font-size: 16px;">Failed</h3>
        ${data.failedPlatforms.map(p => `
          <div style="margin-bottom: 16px; padding: 12px; background: #fff5f5; border-radius: 6px; border-left: 4px solid #cf222e;">
            <strong style="color: #24292f;">${escapeHtml(p.name)}</strong>
            <p style="margin: 4px 0 12px; color: #57606a; font-size: 14px;">${escapeHtml(p.error)}</p>
            <a href="${escapeHtml(p.retryLink)}"
               style="display: inline-block; background-color: #0969da; color: #ffffff; padding: 8px 16px; text-decoration: none; border-radius: 6px; font-size: 14px;">
              Retry ${escapeHtml(p.name)}
            </a>
          </div>
        `).join('')}
      </td>
    </tr>
  `;

  return `<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
</head>
<body style="font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Helvetica, Arial, sans-serif; margin: 0; padding: 0; background-color: #f6f8fa;">
  <table role="presentation" style="width: 100%; border-collapse: collapse;">
    <tr>
      <td align="center" style="padding: 40px 20px;">

          <tr>
            <td style="padding: 32px 32px 16px;">
              <h1 style="margin: 0; font-size: 24px; font-weight: 600; color: #24292f;">
                ${data.successfulPlatforms.length > 0 ? 'Some platforms failed' : 'Posting failed'}
              </h1>
              <p style="margin: 8px 0 0; font-size: 14px; color: #57606a;">
                Post: ${escapeHtml(data.postTitle)}
              </p>
            </td>
          </tr>
          ${successSection}
          ${failedSection}
          <tr>
            <td style="padding: 24px 32px 32px; border-top: 1px solid #d0d7de; background-color: #f6f8fa;">
              <p style="margin: 0; font-size: 14px; color: #57606a;">
                Retry links expire in ${data.expirationHours} hours.
              </p>
            </td>
          </tr>
        </table>
      </td>
    </tr>
  </table>
</body>
</html>`;
}

function renderFailureText(data: FailureNotificationData): string {
  const lines = [
    data.successfulPlatforms.length > 0 ? 'SOME PLATFORMS FAILED' : 'POSTING FAILED',
    '='.repeat(50),
    '',
    `Post: ${data.postTitle}`,
    ''
  ];

  if (data.successfulPlatforms.length > 0) {
    lines.push('SUCCEEDED', '-'.repeat(20), '');
    for (const p of data.successfulPlatforms) {
      lines.push(`- ${p.name}${p.url ? `: ${p.url}` : ''}`);
    }
    lines.push('');
  }

  lines.push('FAILED', '-'.repeat(20), '');
  for (const p of data.failedPlatforms) {
    lines.push(`- ${p.name}`);
    lines.push(`  Error: ${p.error}`);
    lines.push(`  Retry: ${p.retryLink}`);
    lines.push('');
  }

  lines.push(
    '-'.repeat(50),
    '',
    `Retry links expire in ${data.expirationHours} hours.`
  );

  return lines.join('\n');
}

// Send failure notification if any platforms failed
if (summary.failedPlatforms.length > 0) {
  console.log('Sending failure notification email...');

  const config = await loadConfig();
  const emailProvider = createEmailProvider(config.email);
  const secret = process.env.HMAC_SECRET || '';
  const approvalUrl = process.env.APPROVAL_URL || '';

  const failedPlatformsData = summary.results
    .filter(r => !r.success)
    .map(r => {
      const retryToken = generateRetryToken(postId, [r.platform], config.approval.expirationHours, secret);
      return {
        name: r.platform,
        error: r.error || 'Unknown error',
        retryLink: `${approvalUrl}?token=${retryToken}`
      };
    });

  const successfulPlatformsData = summary.results
    .filter(r => r.success)
    .map(r => ({
      name: r.platform,
      url: r.postUrl
    }));

  const emailData = {
    postTitle: post.digest?.title || 'Untitled Post',
    successfulPlatforms: successfulPlatformsData,
    failedPlatforms: failedPlatformsData,
    expirationHours: config.approval.expirationHours
  };

  const template = renderFailureNotificationEmail(emailData);

  try {
    await emailProvider.send({
      to: process.env.NOTIFICATION_EMAIL || config.email.fromEmail,
      subject: template.subject,
      html: template.html,
      text: template.text
    });
    console.log('Failure notification sent');
  } catch (emailError) {
    console.error('Failed to send notification email:', emailError);
    // Don't fail the action - platform posting results are already saved
  }
}
```

Environment variables used:

- APPROVAL_URL: Base URL for retry endpoint
