  <behavior>
GhostPlugin implements PlatformPlugin interface:

isConfigured():
- Returns true when both url and apiKey are truthy
- Returns false when either is missing/empty
- Cases: (url, key) -> true, (url, null) -> false, (null, key) -> false

post(state):
- Returns { success: false, error: "No digest content" } when state.digest missing
- Returns { success: false, error: "Ghost not configured" } when not configured
- Returns { success: true, platformPostId, platformUrl } on successful API call
- Returns { success: false, error: message } on API failure after retries
- Retries up to 3 times on 429 with exponential backoff + jitter
- Does NOT throw exceptions - always returns PostResult

API call structure:
- api.posts.add({ title, html, status, custom_excerpt, tags }, { source: 'html' })
- status MUST be inside post object per RESEARCH.md pitfall #1

```typescript
import GhostAdminAPI from '@tryghost/admin-api';
import type { PlatformPlugin, PostResult } from '../types.js';
import type { PostState } from '../../types/state.js';

const MAX_RETRIES = 3;

export class GhostPlugin implements PlatformPlugin {
  readonly name = 'ghost';
  private api: GhostAdminAPI | null = null;

  constructor(
    private url: string | undefined,
    private apiKey: string | undefined,
    private defaultTags: string[] = ['devlog', 'opensource'],
    private defaultStatus: 'draft' | 'published' = 'draft'
  ) {}

  isConfigured(): boolean {
    return Boolean(this.url && this.apiKey);
  }

  private getClient(): GhostAdminAPI {
    if (!this.api) {
      if (!this.url || !this.apiKey) {
        throw new Error('Ghost not configured');
      }
      this.api = new GhostAdminAPI({
        url: this.url,
        key: this.apiKey,
        version: 'v5.0'
      });
    }
    return this.api;
  }

  async post(state: PostState): Promise<PostResult> {
    if (!state.digest) {
      return { success: false, error: 'No digest content' };
    }

    if (!this.isConfigured()) {
      return { success: false, error: 'Ghost not configured' };
    }

    let attempt = 0;

    while (attempt < MAX_RETRIES) {
      try {
        const api = this.getClient();
        const post = await api.posts.add({
          title: state.digest.title,
          html: state.digest.content,
          status: this.defaultStatus,
          custom_excerpt: state.teaser?.text,
          tags: this.defaultTags.map(name => ({ name }))
        }, { source: 'html' });

        return {
          success: true,
          platformPostId: post.id,
          platformUrl: post.url
        };
      } catch (error: unknown) {
        const apiError = error as { status?: number; message?: string };

        if (apiError?.status === 429 && attempt < MAX_RETRIES - 1) {
          const baseDelay = Math.pow(2, attempt) * 1000;
          const jitter = Math.random() * 1000;
          await new Promise(r => setTimeout(r, baseDelay + jitter));
          attempt++;
          continue;
        }

        return {
          success: false,
          error: apiError?.message || String(error)
        };
      }
    }

    return { success: false, error: 'Max retries exceeded' };
  }
}
```
