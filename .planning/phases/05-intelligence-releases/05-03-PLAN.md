```typescript
export interface ReleasePayload {
  id: number;
  tag_name: string;
  name: string | null;
  body: string | null;
  html_url: string;
  draft: boolean;
  prerelease: boolean;
  published_at: string | null;
  assets: Array<{
    name: string;
    browser_download_url: string;
    size: number;
  }>;
}

export interface ReleaseContentResult {
  announcement: ReleaseAnnouncement;  // For state storage
  postContent: string;                 // Full announcement for Ghost
  teaser: {
    text: string;
    hashtags: string[];
    characterCount: number;
  };
  usage: {
    inputTokens: number;
    outputTokens: number;
  };
}
```
